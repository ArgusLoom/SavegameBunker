```csharp
// ChangeSaveFolderDialogViewModel.cs and AddGameDialogViewModel.cs, identical except the
// missing-folder text (Add uses its cached static FolderMissingStatus field, Change calls
// Strings.Get("str_common_folderNotFound") inline — same string):
private static string ResolveFolder(string folder)
{
    if (!PathResolver.TryResolve(folder, null, out var dir, out var error)) return error!;
    if (!Directory.Exists(dir)) return Strings.Get("str_common_folderNotFound");
    var (count, bytes) = DirectoryWalk.CountFiles(dir);
    return count == 1
        ? Strings.Format("str_common_fileCountOne", SizeFormat.Describe(bytes))
        : Strings.Format("str_common_fileCount", count, SizeFormat.Describe(bytes));
}
[RelayCommand]
private Task BrowseAsync()
{
    var picked = _dialog.PickFolder(Strings.Get("str_common_browseFolderTitle"), null);
    if (picked is not null) SaveFolder = PathResolver.ToTemplate(picked);
    return Task.CompletedTask;
}
```
